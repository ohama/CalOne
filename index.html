Open browser console to test calculator API:

// Test basic arithmetic
inputDigit('5'); updateDisplay();
handleOperator('+');
inputDigit('3'); updateDisplay();
handleOperator('='); updateDisplay();

// Test decimal precision
resetCalculator(); updateDisplay();
inputDigit('0'); inputDecimal(); inputDigit('1');
handleOperator('+');
inputDigit('0'); inputDecimal(); inputDigit('2');
handleOperator('='); updateDisplay();

// Test division by zero
resetCalculator(); updateDisplay();
inputDigit('5'); handleOperator('÷'); inputDigit('0');
handleOperator('='); updateDisplay();

updateDisplay() function updates the display above.
